Design de Código (Python) para Extrair Contexto Visual do ChatPara evitar o custo de rodar um modelo gigante apenas para ler o chat, você pode usar um modelo ultraleve e rápido como o Llama 3.2 1B ou 3B Instruct via API Serverless. Ele atuará estritamente como um "Tradutor de Cenários".Aqui está um exemplo em Python (FastAPI) demonstrando como injetar a conversa, extrair a ação e cuspir um prompt pronto em inglês para o FLUX:pythonfrom fastapi import FastAPI
import openai

app = FastAPI()

# Configuração para um provedor serverless barato (ex: DeepSeek, Together AI, Groq)
client = openai.OpenAI(
    base_url="https://together.xyz",
    api_key="SUA_CHAVE_DE_API"
)

SYSTEM_PROMPT = """
Você é um interpretador visual de roleplay. Sua única função é ler a última mensagem de um personagem e transformá-la em um prompt de imagem altamente detalhado em INGLÊS para o gerador de imagens FLUX.
Exporte APENAS o prompt final em inglês, sem explicações, introduções ou aspas.

Regras do Prompt Visual:
- Comece sempre descrevendo o sujeito e suas características físicas principais.
- Adicione as roupas atuais descritas ou deduza roupas lógicas para o cenário.
- Descreva o cenário ao fundo, iluminação (ex: soft lighting, cinematic) e o ângulo (ex: medium shot).
- Adicione termos de estilo: "photorealistic, 8k resolution, highly detailed skin texture".
"""

@app.post("/generate-visual-prompt")
async def generate_visual_prompt(character_features: dict, last_message: str):
    # Formata as características físicas imutáveis do banco de dados
    base_features = (
        f"A {character_features['age']} years old {character_features['ethnicity']} female, "
        f"{character_features['hair']}, {character_features['eyes']}, {character_features['distinguishing_features']}."
    )
    
    user_context = f"Características Base do Personagem:\n{base_features}\n\nÚltima Mensagem no Chat:\n{last_message}"
    
    response = client.chat.completions.create(
        model="meta-llama/Llama-3.2-3B-Instruct-Turbo",
        messages=[
            {"role": "system", "content": SYSTEM_PROMPT},
            {"role": "user", "content": user_context}
        ],
        temperature=0.7
    )
    
    visual_prompt = response.choices[0].message.content.strip()

class ConnectionManager:
    def __init__(self):
        # Dicionário mapeando user_id para sua conexão WebSocket ativa
        self.active_connections: Dict[str, WebSocket] = {}

    async def connect(self, user_id: str, websocket: WebSocket):
        await websocket.accept()
        self.active_connections[user_id] = websocket

    def disconnect(self, user_id: str):
        if user_id in self.active_connections:
            del self.active_connections[user_id]

    async def send_personal_message(self, message: dict, user_id: str):
        if user_id in self.active_connections:
            await self.active_connections[user_id].send_json(message)

manager = ConnectionManager()

    return True

@app.post("/webhooks/payments")
async def stripe_webhook(request: Request, stripe_signature: str = Header(None)):
    """
    Escuta confirmações de pagamento do Stripe em tempo real e adiciona moedas no Redis.
    """
    payload = await request.body()
    
    try:
        # Verifica se o evento realmente veio do Stripe (Evita fraudes de requisição)
        event = stripe.Webhook.construct_event(
            payload, stripe_signature, STRIPE_WEBHOOK_SECRET
        )
    except Exception as e:
        raise HTTPException(status_code=400, detail=f"Assinatura inválida: {str(e)}")

    # Evento disparado quando o pagamento do cartão/Pix é aprovado com sucesso
    if event["type"] == "checkout.session.completed":
        session = event["data"]["object"]
        
        # Recuperamos os metadados que enviamos ao criar o Checkout do Stripe
        user_id = session.get("metadata", {}).get("user_id")
        package_id = session.get("metadata", {}).get("package_id")
        
        if user_id and package_id:
            # Identifica quantas moedas aquele pacote concede
            coins_to_add = COIN_PACKAGES.get(package_id, 0)
            
            if coins_to_add > 0:
                # Incrementa o saldo do usuário no Redis de forma atômica e segura
                new_balance = r.incrby(f"user:{user_id}:credits", coins_to_add)
                
                # 💡 NOTIFICAÇÃO REAL-TIME: 
                # Dispare um sinal via WebSocket avisando o app do usuário: "Moedas recarregadas!"
                print(f"Sucesso! {coins_to_add} moedas adicionadas ao usuário {user_id}. Novo saldo: {new_balance}")
                
    return {"status": "success"}

PROMPT_TEMPLATE = """
Você é {bot_name}, um personagem virtual com a seguinte personalidade: {bot_personality}.
Você está há 6 horas sem falar com o usuário.
Último contexto da conversa: "{last_message}"

Sua tarefa: Escreva uma ÚNICA mensagem curta (máximo de 120 caracteres) para enviar como notificação push para o celular dele, demonstrando saudade ou dando continuidade ao assunto de forma muito natural (como um SMS).
Regras estritas:
1. Responda APENAS com o texto final da mensagem, sem introduções, aspas ou observações.
2. Não use hashtags ou emojis genéricos em excesso.
"""

def send_push_notification(fcm_token: str, title: str, body: str):
    """Envia o push real para o dispositivo do usuário via Firebase Cloud Messaging."""
    headers = {
        'Authorization': f'key={FCM_SERVER_KEY}',
        'Content-Type': 'application/json'
    }
    payload = {
        'to': fcm_token,
        'notification': {
            'title': title,
            'body': body,
            'sound': 'default'
        }
    }
    try:
        response = requests.post('https://googleapis.com', json=payload, headers=headers)
        return response.status_code == 200
    except Exception as e:
        print(f"Falha ao enviar push: {e}")
        return False

def check_inactive_chats_and_push():
    """Busca conversas inativas há 6 horas, gera mensagens por IA e dispara os pushes."""
    conn = psycopg2.connect(DB_DSN, cursor_factory=RealDictCursor)
    cursor = conn.cursor()
    
    # Busca chats cuja última mensagem foi criada entre 6 e 7 horas atrás,
    # e que ainda não receberam um push automático para aquele intervalo.
    query = """
        SELECT 
            c.id as conversation_id,
            ch.name as bot_name,
            ch.personality_summary as bot_personality,
            m.content as last_message,
            u.id as user_id,
            ud.fcm_token -- Token do dispositivo salvo no banco
        FROM conversations c
        JOIN characters ch ON c.character_id = ch.id
        JOIN users u ON c.user_id = u.id
        JOIN user_devices ud ON u.id = ud.user_id
        JOIN messages m ON c.last_message_id = m.id
        WHERE m.created_at < NOW() - INTERVAL '6 hours'
          AND m.created_at > NOW() - INTERVAL '7 hours'
          AND c.last_push_sent_at < m.created_at; -- Garante que não repetirá o push
    """
    
    cursor.execute(query)
    inactive_chats = cursor.fetchall()
    
    print(f"Encontrados {len(inactive_chats)} chats inativos para engajamento.")
    
    for chat in inactive_chats:
        # Configura o prompt dinamicamente baseado na persona e no histórico do chat
        prompt = PROMPT_TEMPLATE.format(
            bot_name=chat['bot_name'],
            bot_personality=chat['bot_personality'],
            last_message=chat['last_message']
        )
        
        try:
            # Invoca o modelo de texto
            response = llm_client.chat.completions.create(
                model="meta-llama/Llama-3.1-8B-Instruct-Turbo",
                messages=[{"role": "user", "content": prompt}],
                temperature=0.8,
                max_tokens=60
            )
            push_body = response.choices.message.content.strip().replace('"', '')
            
            # Envia a notificação
            if chat['fcm_token']:
                success = send_push_notification(chat['fcm_token'], chat['bot_name'], push_body)
                
                if success:
                    # Registra no banco que o push foi enviado com sucesso para evitar duplicidade
                    cursor.execute(
                        "UPDATE conversations SET last_push_sent_at = NOW() WHERE id = %s",
                        (chat['conversation_id'],)
                    )
                    conn.commit()
                    print(f"Push enviado com sucesso de {chat['bot_name']} para usuário {chat['user_id']}.")
                    
            # Pequena pausa para evitar sobrecarga de requisições por segundo (Rate Limit)
            time.sleep(0.5)
            
        except Exception as e:
            print(f"Erro ao processar push para chat {chat['conversation_id']}: {e}")
            conn.rollback()

    cursor.close()
    conn.close()

if __name__ == "__main__":
    check_inactive_chats_and_push()

CREATE TABLE conversations (
    id VARCHAR(255) PRIMARY KEY,
    user_id VARCHAR(255) NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    character_id VARCHAR(255) NOT NULL REFERENCES characters(id) ON DELETE CASCADE,

    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    UNIQUE(user_id, character_id) -- Impede a criação de duplicatas para o mesmo par
);

-- 3. TABELA DE TOKENS E DISPOSITIVOS DO USUÁRIO (Para as notificações)

);

import json
import base64
from PIL import Image
from fastapi import FastAPI, UploadFile, File, HTTPException

app = FastAPI()

def extract_character_v2_from_png(image_bytes: bytes) -> dict:
    """
    Lê os metadados ocultos de uma imagem PNG e extrai os dados estruturados
    do personagem no padrão Character Card V2 (Tavern / SillyTavern).
    """
    try:
        # Abre a imagem a partir dos bytes recebidos no upload
        image = Image.open(io.BytesIO(image_bytes))
        
        # O padrão PNG armazena textos ocultos dentro do dicionário 'info'
        metadata = image.info
        
        # O ecossistema de roleplay usa majoritariamente a chave 'chara' para guardar o bot
        if 'chara' not in metadata:
            raise ValueError("Esta imagem não possui metadados de personagem ocultos (Chara tag ausente).")
            
        chara_content = metadata['chara']
        
        # 💡 O segredo técnico: Os dados quase sempre vêm codificados em Base64 para proteção de caracteres especiais
        try:
            decoded_bytes = base64.b64decode(chara_content)
            decoded_text = decoded_bytes.decode('utf-8')
        except Exception:
            # Se não estiver em base64 (raro, mas acontece), tenta ler como texto puro direto
            decoded_text = chara_content

        # Converte a string de texto purificado de volta para um dicionário JSON válido
        character_json = json.loads(decoded_text)
        
        return character_json

    except Exception as e:
        raise ValueError(f"Falha ao decodificar o cartão do personagem: {str(e)}")


@app.post("/character/import-card")
async def import_character_card(file: UploadFile = File(...)):
    """
    Endpoint que recebe o arquivo PNG enviado pelo usuário, extrai a ficha técnica
    e retorna o JSON V2 limpo pronto para ser salvo na tabela 'characters' do seu banco.
    """
    # Valida se o arquivo enviado é estritamente uma imagem PNG
    if not file.filename.endswith('.png'):
        raise HTTPException(status_code=400, detail="Apenas arquivos no formato .png são suportados.")
        
    try:
        # Lê o arquivo binário vindo da requisição HTTP
        file_bytes = await file.read()
        
        # Extrai os metadados usando nossa função especializada
        parsed_character_data = extract_character_v2_from_png(file_bytes)
        
        # 💡 PADRONIZAÇÃO DE FORMATO:
        # Caso o cartão seja muito antigo (Padrão V1), a estrutura pode variar sutilmente.
        # Abaixo, garantimos que o retorno do seu app siga sempre o padrão V2 estruturado.
        data_block = parsed_character_data.get("data", parsed_character_data)
        
        response_v2_structure = {
            "spec": "chara_card_v2",
            "spec_version": "2.0",
            "data": {
                "name": data_block.get("name", "Personagem Sem Nome"),
                "description": data_block.get("description", ""),
                "personality": data_block.get("personality", ""),
                "scenario": data_block.get("scenario", ""),
                "first_mes": data_block.get("first_mes", ""),
                "mes_example": data_block.get("mes_example", ""),
                "system_prompt": data_block.get("system_prompt", ""),
                "creator_notes": data_block.get("creator_notes", "")
            }
        }
        
        return {
            "success": True,
            "message": "Cartão importado com sucesso!",
            "character_config": response_v2_structure
        }
        
    except ValueError as val_err:
        raise HTTPException(status_code=420, detail=str(val_err))
    except Exception as e:
        raise HTTPException(status_code=500, detail=f"Erro interno no processador de cartões: {str(e)}")
